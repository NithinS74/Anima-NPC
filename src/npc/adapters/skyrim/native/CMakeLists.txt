cmake_minimum_required(VERSION 3.20)
project(AnimaConnector VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "AnimaConnector requires Windows x64")
endif()
set(ANIMA_DEPS "${CMAKE_CURRENT_LIST_DIR}/../../../../../.tools" CACHE PATH "Pinned local checkouts")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SPDLOG_INSTALL ON CACHE BOOL "" FORCE)
set(SPDLOG_FMT_EXTERNAL ON CACHE BOOL "" FORCE)
set(FMT_TEST OFF CACHE BOOL "" FORCE)
set(FMT_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory("${ANIMA_DEPS}/fmt" "${CMAKE_BINARY_DIR}/fmt")
set(fmt_DIR "${CMAKE_CURRENT_LIST_DIR}/cmake")
add_subdirectory("${ANIMA_DEPS}/spdlog" "${CMAKE_BINARY_DIR}/spdlog")
set(spdlog_DIR "${CMAKE_CURRENT_LIST_DIR}/cmake")
set(RAPIDCSV_INCLUDE_DIRS "${ANIMA_DEPS}/rapidcsv/src" CACHE PATH "" FORCE)
add_subdirectory("${ANIMA_DEPS}/CommonLibSSE-NG" "${CMAKE_BINARY_DIR}/commonlib")
add_library(AnimaConnector SHARED plugin.cpp)
target_include_directories(AnimaConnector PRIVATE "${ANIMA_DEPS}/json/single_include" "${RAPIDCSV_INCLUDE_DIRS}")
target_link_libraries(AnimaConnector PRIVATE CommonLibSSE::CommonLibSSE winhttp)
target_precompile_headers(AnimaConnector PRIVATE <RE/Skyrim.h> <SKSE/SKSE.h>)
target_compile_definitions(AnimaConnector PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(AnimaConnector PRIVATE /utf-8 /EHsc /W4)
install(TARGETS AnimaConnector RUNTIME DESTINATION SKSE/Plugins LIBRARY DESTINATION SKSE/Plugins)
install(FILES AnimaConnector.ini DESTINATION SKSE/Plugins)
enable_testing()
add_executable(AnimaDeliveryTests "${CMAKE_CURRENT_LIST_DIR}/../../../../../tests/unit/adapters/skyrim_delivery.cpp")
target_include_directories(AnimaDeliveryTests PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
add_test(NAME skyrim_delivery COMMAND AnimaDeliveryTests)
